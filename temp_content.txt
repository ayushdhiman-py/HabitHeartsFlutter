import 'dart:math' as math;

class GoalProgressSummary {
  final Map<String, String> monthlyData; // YYYY-MM -> bit string
  final int currentStreak;
  final int longestStreak;
  final DateTime lastUpdated;

  GoalProgressSummary({
    this.monthlyData = const {},
    this.currentStreak = 0,
    this.longestStreak = 0,
    required this.lastUpdated,
  });

  Map<String, dynamic> toJson() {
    return {
      'monthlyData': monthlyData,
      'currentStreak': currentStreak,
      'longestStreak': longestStreak,
      'lastUpdated': lastUpdated.millisecondsSinceEpoch,
    };
  }

  factory GoalProgressSummary.fromJson(Map<String, dynamic> json) {
    return GoalProgressSummary(
      monthlyData: Map<String, String>.from(json['monthlyData'] ?? {}),
      currentStreak: json['currentStreak'] ?? 0,
      longestStreak: json['longestStreak'] ?? 0,
      lastUpdated: DateTime.fromMillisecondsSinceEpoch(
          json['lastUpdated'] is int
              ? json['lastUpdated']
              : (json['lastUpdated'] as Map<String, dynamic>).containsKey('_seconds')
                  ? json['lastUpdated']['_seconds'] * 1000
                  : json['lastUpdated']['seconds'] * 1000),
    );
  }

  // Helper methods for bit manipulation
  bool isDayCompleted(String yearMonth, int day) {
    if (!monthlyData.containsKey(yearMonth)) return false;
    
    String bitString = monthlyData[yearMonth]!;
    if (day < 1 || day > bitString.length) return false;
    
    int index = day - 1;
    return index < bitString.length && bitString[index] == '1';
  }

  String setDayCompleted(String yearMonth, int day, bool completed, int daysInMonth) {
    // Get existing bit string or create new one
    String bitString = monthlyData[yearMonth] ?? '0' * daysInMonth;
    
    // Ensure bit string has correct length
    if (bitString.length < daysInMonth) {
      bitString = bitString.padRight(daysInMonth, '0');
    } else if (bitString.length > daysInMonth) {
      bitString = bitString.substring(0, daysInMonth);
    }
    
    // Update the specific day
    if (day >= 1 && day <= bitString.length) {
      int index = day - 1;
      List<String> bits = bitString.split('');
      bits[index] = completed ? '1' : '0';
      bitString = bits.join('');
    }
    
    return bitString;
  }

  List<int> getCompletedDays(String yearMonth) {
    if (!monthlyData.containsKey(yearMonth)) return [];
    
    String bitString = monthlyData[yearMonth]!;
    List<int> completedDays = [];
    
    for (int i = 0; i < bitString.length; i++) {
      if (bitString[i] == '1') {
        completedDays.add(i + 1); // Convert to 1-based day
      }
    }
    
    return completedDays;
  }

  // Calculate current streak (counting backwards from today)
  int getCurrentStreak() {
    int currentStreak = 0;
    DateTime currentDate = DateTime.now();
    bool continueCounting = true;
    
    while (continueCounting) {
      String currentYearMonth = '${currentDate.year}-${currentDate.month.toString().padLeft(2, '0')}';
      int currentDay = currentDate.day;
      
      // Check if we have data for this month
      if (!monthlyData.containsKey(currentYearMonth)) {
        // If no data for this month, break the streak counting
        break;
      }
      
      String bitString = monthlyData[currentYearMonth]!;
      int index = currentDay - 1;
      
      // Check if index is valid and day exists in bitString
      if (index >= 0 && index < bitString.length) {
        if (bitString[index] == '1') {
          // Day is completed, increment streak
          currentStreak++;
          // Move to previous day
          currentDate = DateTime(currentDate.year, currentDate.month, currentDate.day - 1);
        } else {
          // Day is not completed, stop the streak counting
          continueCounting = false;
        }
      } else {
        // Day doesn't exist in this month, go to previous day
        currentDate = DateTime(currentDate.year, currentDate.month, currentDate.day - 1);
      }
      
      // Check if we've gone beyond a reasonable date range (e.g., more than 2 years back)
      DateTime twoYearsBack = DateTime.now().subtract(Duration(days: 730)); // Approx 2 years
      if (currentDate.isBefore(twoYearsBack)) {
        break;
      }
    }
    
    return currentStreak;
  }

  // Calculate longest streak across all months
  int getLongestStreak() {
    int longestStreak = 0;
    int currentPotentialStreak = 0;
    bool inStreak = false;
    
    // Get all months in chronological order
    List<String> months = monthlyData.keys.toList()..sort();
    
    // Process each month to calculate longest streak
    for (String month in months) {
      String? bitString = monthlyData[month];
      if (bitString == null) continue;
      
      // For each day in this month
      for (int i = 0; i < bitString.length; i++) {
        if (bitString[i] == '1') {
          // Day is completed, increment streak
          currentPotentialStreak++;
          inStreak = true;
        } else {
          // Day is not completed, reset streak counter
          if (inStreak) {
            longestStreak = math.max(longestStreak, currentPotentialStreak);
            currentPotentialStreak = 0;
            inStreak = false;
          }
        }
      }
    }
    
    // Don't forget to check the final potential streak if it was ongoing
    if (inStreak) {
      longestStreak = math.max(longestStreak, currentPotentialStreak);
    }
    
    // Also consider the stored longest streak value
    longestStreak = math.max(longestStreak, this.longestStreak);
